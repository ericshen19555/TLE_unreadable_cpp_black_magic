## 稀疏表

```cpp
template <typename T, auto f>
struct ST {
    static constexpr int M = __lg(N) + 1;
    T st[M][N];

    void build(T *l, int n) {
        const int lim = n + 1;
        int bl = __lg(n) + 1;

        REP(i, 0, n) st[0][i] = l[i];
        int b = 1;
        REP(r, 1, bl) {
            REP(i, 0, n - b) {
                st[r][i] = f(st[r - 1][i], st[r - 1][i + b]);
            }
            b <<= 1;
        }
    }
    T query(int s, int t) {
        int r = __lg(t - s);
        return f(st[r][s], st[r][t - (1 << r)]);
    }
};
```

```cpp
void update(int i, int v, int h) {
    zkw[i] += v << h;

void push(int i) {
    for (int h = __lg(i); h; ) {
        int j = i >> h--;
        if (tag[j]) {
            update(j << 1, tag[j], h);
            update(j << 1 | 1, tag[j], h);
            tag[j] = 0;
        }
    }
}
void pull(int i) {
    for (int h = 1; i >>= 1; ++h) {
        zkw[i] = zkw[i << 1] + zkw[i << 1 | 1] + (tag[i] << h)
    }
}
```

## 懶標遞迴式線段樹

```cpp
ll l[N];
struct node {
    ll val, tag;
} tr[N << 2];

node combine(node a, node b) {
    // combine
    return {a.val + b.val, 0};
}

void merge(int pos, int s, int t) {
    int mid = (s + t) >> 1;
    tr[pos] = combine(tr[pos << 1], tr[pos << 1 | 1]);
}

// init
void build(int pos, int s, int t) {
    // init tag
    tr[pos].tag = 0;
    if (s + 1 == t) {
        // init leaf
        tr[pos].val = l[s];
        return;
    }
    int mid = (s + t) >> 1;
    build(pos << 1 | 0, s, mid);
    build(pos << 1 | 1, mid, t);
    merge(pos, s, t);
}

void update(int pos, int s, int t, node o) {
    int mid = (s + t) >> 1;
    // update tree node
    tr[pos].val += o.tag * (t - s);
    // update tag
    tr[pos].tag += o.tag;
}

void push(int pos, int s, int t) {
    if (s + 1 == t) return;
    int mid = (s + t) >> 1;
    // push tag
    update(pos << 1 | 0, s, mid, tr[pos]);
    update(pos << 1 | 1, mid, t, tr[pos]);
    // reset tag
    tr[pos].tag = 0;
}

node query(int pos, int s, int t, int qs, int qt) {
    if (qt <= s or  t <= qs) return {0, 0};  // empty
    if (qs <= s and t <= qt) return tr[pos];
    push(pos, s, t);
    int mid = (s + t) >> 1;
    auto le = query(pos << 1 | 0, s, mid, qs, qt);
    auto ri = query(pos << 1 | 1, mid, t, qs, qt);
    return combine(le, ri);
}

void modify(int pos, int s, int t, int qs, int qt, node o) {
    if (qt <= s or  t <= qs) return;
    if (qs <= s and t <= qt) {
        // place tag
        update(pos, s, t, o);
        return;
    }
    push(pos, s, t);
    int mid = (s + t) >> 1;
    modify(pos << 1 | 0, s, mid, qs, qt, o);
    modify(pos << 1 | 1, mid, t, qs, qt, o);
    merge(pos, s, t);
}
```

## FHQ-Treap

TLE / MLE: 檢查 `node.siz`、`siz(o)`、`rng` 是否 `long long`
WA：`update` 前需要 `clone`
WA：如果 `merge` 前沒有 `split` 則需要 `clone`
WA：所有地方都加上 `push`

```cpp
mt19937 rng(114514);
// mt19937_64 rng(1145141919810);

struct node {
    int val, siz;
    node *le, *ri;
    node(int v): val(v), siz(1), le(0), ri(0) {}
    void push() {}
    void pull() {
        siz = 1;
        if (le) siz += le->siz;
        if (ri) siz += ri->siz;
    }
    node *clone() {
        // clone before update!!!
        return this;
        // return new node(*this);  // persistent
    }
};

auto siz(node *x) {
    return x ? x->siz : 0;
}

node *merge(node *a, node *b) {
    if (not a or not b) return a ? a : b;
    // if no split before merge, need clone
    if (rng() % (siz(a) + siz(b)) < siz(a)) {
        a->push(), a->ri = merge(a->ri, b), a->pull();
        return a;
    } else {
        b->push(), b->le = merge(a, b->le), b->pull();
        return b;
    }
}

void split(node *o, node *&a, node *&b, int val) {
    if (not o) {
        a = b = 0; return;
    }
    o = o->clone();
    o->push();
    if (o->val < val) {
        a = o, split(o->ri, a->ri, b, val);
    } else {
        b = o, split(o->le, a, b->le, val);
    }
    o->pull();
}

void split2(node *o, node *&a, node *&b, int k) {
    if (not o) {
        a = b = 0; return;
    }
    o = o->clone();
    o->push();
    if (siz(o->le) < k) {
        a = o, split2(o->ri, a->ri, b, k - siz(o->le) - 1);
    } else {
        b = o, split2(o->le, a, b->le, k);
    }
    o->pull();
}

int rk(node *o, int val) {
    if (not o) return 0;
    if (o->val < val) {
        return siz(o->le) + 1 + rk(o->ri, val);
    } else {
        return rk(o->le, val);
    }
}

int kth(node *o, int k) {
    if (siz(o->le) < k) return kth(o->ri, k - siz(o->le) - 1);
    if (siz(o->le) == k) return o->val;
    return kth(o->le, k);
}

void insert(node *&o, int val) {
    node *a, *b;
    split(o, a, b, val);
    o = merge(a, merge(new node(val), b));
}

void remove(node *&o, int val) {
    node *a, *b, *c;
    split(o, a, b, val), split2(b, b, c, 1);
    if (b and b->val == val) delete b, b = 0;
    o = merge(a, merge(b, c));
}

node *meld(node *a, node *b) {
    if (not a or not b) return a ? a : b;
    if (rng() % (siz(a) + siz(b)) >= siz(a)) swap(a, b);
    a->push();
    node *c; split(b, b, c, a->val);
    a->le = meld(a->le, b);
    a->ri = meld(a->ri, c);
    a->pull();
    return a;
}
```

## 配對堆

```cpp
struct node {
    ll val; node *ch, *si;
    node(ll v): val(v), ch(0), si(0) {}
};

node *meld(node *a, node *b) {
    if (not a or not b) return a ? a : b;
    if (a == b) return a;
    if (a->val > b->val) swap(a, b);
    b->si = a->ch;
    a->ch = b;
    return a;
}

node *merges(node *a) {
    if (not a or not a->si) return a;
    node *b = a->si, *c = b->si;
    a->si = b->si = 0;
    return meld(merges(c), meld(a, b));
}

ll top(node *o) {
    return o ? o->val : -1;
}

void pop(node *&o) {
    if (not o) return;
    node *t = o;
    o = merges(o->ch);
    delete t;
}
```

## 隨機堆

```cpp
struct node {
    ll val; node *le, *ri;
    node(ll v): val(v), le(0), ri(0) {}
    node *clone() {
        return this;
        // return new node(*this)  # persistent
    }
};

node *merge(node *a, node *b) {
    if (not a or not b) return a ? a : b;
    if (a == b) return a;
    if (a->val > b->val) swap(a, b);
    a = a->clone();
    a->ri = merge(a->ri, b);
    if (rand() & 1) swap(a->le, a->ri);
    return a;
}

ll top(node *o) {
    return o ? o->val : -1;
}

void pop(node *&o) {
    if (not o) return;
    node *t = o;
    o = merge(o->le, o->ri);
    delete t;
}

node *pop(node *o) {  // persistent
    if (not o) return 0;
    node *t = merge(o->le, o->ri);
    delete o;
    return t;
}
```

## 李超樹

`update` 往下傳 `v`，而非 `u`。

```cpp
struct line {
    ll m = 0, k = 0;
    ll y(ll x) {
        return x * m + k;
    }
} tr[N << 2];

void update(int o, int s, int t, line v) {
    auto &u = tr[o];
    int mid = s + t >> 1;
    if (v.y(mid) > u.y(mid)) swap(u, v);
    if (v.y(s+0) > u.y(s+0)) update(o << 1 | 0, s, mid, v);
    if (v.y(t-1) > u.y(t-1)) update(o << 1 | 1, mid, t, v);
}

void modify(int o, int s, int t, int qs, int qt, line v) {
    if (qt <= s or  t <= qs) return;
    if (qs <= s and t <= qt) {
        update(o, s, t, v);
        return;
    }
    int mid = s + t >> 1;
    modify(o << 1 | 0, s, mid, qs, qt, v);
    modify(o << 1 | 1, mid, t, qs, qt, v);
}

ll query(int o, int s, int t, int x) {
    ll res = tr[o].y(x);
    if (s + 1 == t) return res;
    int mid = s + t >> 1;
    if (x < mid) res = max(res, query(o << 1 | 0, s, mid, x));
    else         res = max(res, query(o << 1 | 1, mid, t, x));
    return res;
}
```

## SortedList

```cpp
template <typename T>
class SortedList {
private:
    static const int DEFAULT_LOAD_FACTOR = 1000;
    int _len;
    int _load;
    std::vector<std::vector<T>> _lists;
    std::vector<T> _maxes;
    std::vector<int> _index;
    void _build_index() {
        _index.assign(_lists.size() + 1, 0);
        for (size_t i = 0; i < _lists.size(); ++i) {
            _index[i + 1] = (int)_lists[i].size();
        }
        int len_index = (int)_index.size();
        for (int i = 1; i < len_index; ++i) {
            int parent = i + (i & -i);
            if (parent < len_index) {
                _index[parent] += _index[i];
            }
        }
    }
    void _clear_content() {
        _len = 0;
        _lists.clear();
        _maxes.clear();
        _index.clear();
    }
    std::pair<int, int> _pos(int idx) {
        if (idx < 0) {
            int last_len = (int)_lists.back().size();
            if (-idx <= last_len) {
                return {(int)_lists.size() - 1, last_len + idx};
            }
            idx += _len;
            if (idx < 0) throw std::out_of_range("list index out of range");
        } else if (idx >= _len) {
            throw std::out_of_range("list index out of range");
        }
        if (idx < (int)_lists[0].size()) {
            return {0, idx};
        }
        if (_index.empty()) _build_index();
        int pos = 0;
        int len_index = (int)_index.size();
        int bit = 1;
        while (bit <= len_index) bit <<= 1;
        bit >>= 1;
        while (bit > 0) {
            int new_pos = pos + bit;
            if (new_pos < len_index && _index[new_pos] <= idx) {
                idx -= _index[new_pos];
                pos = new_pos;
            }
            bit >>= 1;
        }
        return {pos, idx};
    }
    void _expand(int pos) {
        if (_lists[pos].size() > (size_t)(_load << 1)) {
            auto& current_list = _lists[pos];
            int split_idx = _load;
            std::vector<T> new_list;
            new_list.reserve(current_list.size() - split_idx);
            std::move(current_list.begin() + split_idx, current_list.end(), std::back_inserter(new_list));
            current_list.resize(split_idx);
            _maxes[pos] = current_list.back();
            _lists.insert(_lists.begin() + pos + 1, std::move(new_list));
            _maxes.insert(_maxes.begin() + pos + 1, _lists[pos + 1].back());
            _index.clear();
        } else {
            if (!_index.empty()) {
                int len_index = (int)_index.size();
                int ptr = pos + 1;
                while (ptr < len_index) {
                    _index[ptr] += 1;
                    ptr += ptr & -ptr;
                }
            }
        }
    }
    void _delete_item(int pos, int idx) {
        _lists[pos].erase(_lists[pos].begin() + idx);
        _len--;
        int len_lists_pos = (int)_lists[pos].size();
        if (len_lists_pos > (_load >> 1)) {
            _maxes[pos] = _lists[pos].back();
            if (!_index.empty()) {
                int len_index = (int)_index.size();
                int ptr = pos + 1;
                while (ptr < len_index) {
                    _index[ptr] -= 1;
                    ptr += ptr & -ptr;
                }
            }
        } else if (_lists.size() > 1) {
            if (pos == 0) pos += 1;
            int prev = pos - 1;
            _lists[prev].insert(_lists[prev].end(), std::make_move_iterator(_lists[pos].begin()), std::make_move_iterator(_lists[pos].end()));
            _maxes[prev] = _lists[prev].back();
            _lists.erase(_lists.begin() + pos);
            _maxes.erase(_maxes.begin() + pos);
            _index.clear();
            _expand(prev);
        } else if (len_lists_pos > 0) {
            _maxes[pos] = _lists[pos].back();
        } else {
            _lists.erase(_lists.begin() + pos);
            _maxes.erase(_maxes.begin() + pos);
            _index.clear();
        }
    }
    int _loc(int pos, int idx) {
        if (pos == 0) return idx;
        if (_index.empty()) _build_index();
        int total = idx;
        while (pos > 0) {
            total += _index[pos];
            pos &= pos - 1;
        }
        return total;
    }
public:
    SortedList(int load_factor = DEFAULT_LOAD_FACTOR) : _len(0), _load(load_factor) {
        _lists.reserve(16);
        _maxes.reserve(16);
    }
    SortedList(const std::vector<T>& initial_values, int load_factor = DEFAULT_LOAD_FACTOR) : _len(0), _load(load_factor) {
        update(initial_values);
    }
    void clear() {
        _clear_content();
    }
    void add(const T& value) {
        if (_maxes.empty()) {
            _lists.push_back({value});
            _maxes.push_back(value);
            _len++;
            return;
        }
        auto it = std::upper_bound(_maxes.begin(), _maxes.end(), value);
        int pos = (int)(it - _maxes.begin());
        if (pos == (int)_maxes.size()) {
            pos -= 1;
            _lists[pos].push_back(value);
            _maxes[pos] = value;
        } else {
            auto& target_list = _lists[pos];
            auto ins_it = std::upper_bound(target_list.begin(), target_list.end(), value);
            target_list.insert(ins_it, value);
        }
        _len++;
        _expand(pos);
    }
    void update(const std::vector<T>& iterable) {
        if (iterable.empty()) return;
        std::vector<T> values = iterable;
        std::sort(values.begin(), values.end());
        if (!_maxes.empty()) {
            if ((long long)values.size() * 4 >= _len) {
                std::vector<T> all_values;
                all_values.reserve(_len + values.size());
                for (auto& lst : _lists) {
                    all_values.insert(all_values.end(), std::make_move_iterator(lst.begin()), std::make_move_iterator(lst.end()));
                }
                std::vector<T> merged;
                merged.reserve(_len + values.size());
                std::merge(all_values.begin(), all_values.end(), values.begin(), values.end(), std::back_inserter(merged));
                _clear_content();
                values = std::move(merged);
            } else {
                for (const auto& val : values) add(val);
                return;
            }
        }
        _len = (int)values.size();
        _index.clear();
        _lists.clear();
        _maxes.clear();
        for (size_t i = 0; i < values.size(); i += _load) {
            size_t end = std::min(i + _load, values.size());
            std::vector<T> chunk(values.begin() + i, values.begin() + end);
            _maxes.push_back(chunk.back());
            _lists.push_back(std::move(chunk));
        }
    }
    bool contains(const T& value) const {
        if (_maxes.empty()) return false;
        auto it = std::lower_bound(_maxes.begin(), _maxes.end(), value);
        if (it == _maxes.end()) return false;
        int pos = (int)(it - _maxes.begin());
        return std::binary_search(_lists[pos].begin(), _lists[pos].end(), value);
    }
    void discard(const T& value) {
        if (_maxes.empty()) return;
        auto it = std::lower_bound(_maxes.begin(), _maxes.end(), value);
        if (it == _maxes.end()) return;
        int pos = (int)(it - _maxes.begin());
        auto& lst = _lists[pos];
        auto elem_it = std::lower_bound(lst.begin(), lst.end(), value);
        if (elem_it != lst.end() && *elem_it == value) {
            _delete_item(pos, (int)(elem_it - lst.begin()));
        }
    }
    void remove(const T& value) {
        if (_maxes.empty()) throw std::invalid_argument("Value not in list");
        auto it = std::lower_bound(_maxes.begin(), _maxes.end(), value);
        if (it == _maxes.end()) throw std::invalid_argument("Value not in list");
        int pos = (int)(it - _maxes.begin());
        auto& lst = _lists[pos];
        auto elem_it = std::lower_bound(lst.begin(), lst.end(), value);
        if (elem_it != lst.end() && *elem_it == value) {
            _delete_item(pos, (int)(elem_it - lst.begin()));
        } else {
            throw std::invalid_argument("Value not in list");
        }
    }
    T pop(int index = -1) {
        if (_len == 0) throw std::out_of_range("pop index out of range");
        if (index == 0) {
            T val = _lists[0][0];
            _delete_item(0, 0);
            return val;
        }
        if (index == -1) {
            int pos = (int)_lists.size() - 1;
            int loc = (int)_lists[pos].size() - 1;
            T val = _lists[pos][loc];
            _delete_item(pos, loc);
            return val;
        }
        auto p = _pos(index);
        int pos = p.first;
        int idx = p.second;
        T val = _lists[pos][idx];
        _delete_item(pos, idx);
        return val;
    }
    T operator[](int index) {
        if (_len == 0) throw std::out_of_range("list index out of range");
        if (index == 0) return _lists[0][0];
        if (index == -1) return _lists.back().back();
        if (index >= 0 && index < (int)_lists[0].size()) {
            return _lists[0][index];
        }
        if (index < 0) {
            int len_last = (int)_lists.back().size();
            if (index >= -len_last) return _lists.back()[len_last + index];
        }
        auto p = _pos(index);
        return _lists[p.first][p.second];
    }
    int bisect_left(const T& value) {
        if (_maxes.empty()) return 0;
        auto it = std::lower_bound(_maxes.begin(), _maxes.end(), value);
        if (it == _maxes.end()) return _len;
        int pos = (int)(it - _maxes.begin());
        auto elem_it = std::lower_bound(_lists[pos].begin(), _lists[pos].end(), value);
        return _loc(pos, (int)(elem_it - _lists[pos].begin()));
    }
    int bisect_right(const T& value) {
        if (_maxes.empty()) return 0;
        auto it = std::upper_bound(_maxes.begin(), _maxes.end(), value);
        if (it == _maxes.end()) return _len;
        int pos = (int)(it - _maxes.begin());
        auto elem_it = std::upper_bound(_lists[pos].begin(), _lists[pos].end(), value);
        return _loc(pos, (int)(elem_it - _lists[pos].begin()));
    }
    int count(const T& value) {
        if (_maxes.empty()) return 0;
        int pos_left = (int)(std::lower_bound(_maxes.begin(), _maxes.end(), value) - _maxes.begin());
        if (pos_left == (int)_maxes.size()) return 0;
        int idx_left = (int)(std::lower_bound(_lists[pos_left].begin(), _lists[pos_left].end(), value) - _lists[pos_left].begin());
        int pos_right = (int)(std::upper_bound(_maxes.begin(), _maxes.end(), value) - _maxes.begin());
        if (pos_right == (int)_maxes.size()) {
            return _len - _loc(pos_left, idx_left);
        }
        int idx_right = (int)(std::upper_bound(_lists[pos_right].begin(), _lists[pos_right].end(), value) - _lists[pos_right].begin());
        if (pos_left == pos_right) return idx_right - idx_left;
        return _loc(pos_right, idx_right) - _loc(pos_left, idx_left);
    }
    int index(const T& value) {
        if (_maxes.empty()) throw std::invalid_argument("Value not in list");
        int pos = (int)(std::lower_bound(_maxes.begin(), _maxes.end(), value) - _maxes.begin());
        if (pos == (int)_maxes.size()) throw std::invalid_argument("Value not in list");
        auto& lst = _lists[pos];
        auto it = std::lower_bound(lst.begin(), lst.end(), value);
        if (it == lst.end() || *it != value) throw std::invalid_argument("Value not in list");
        return _loc(pos, (int)(it - lst.begin()));
    }
    int size() const { return _len; }
    bool empty() const { return _len == 0; }
};
```

## Boruvka
- 每一輪，每個**連通塊**（以連通塊為單位，而非單點） 向其他連通塊**僅**嘗試**最小的一條邊**，要小心不要一次更新兩邊以上。
- 連通塊的判斷方式，是在所有 merge 開始之前，所以要另外開一個 array 紀錄 merge 前的所屬 group，而不能使用 dsu find 動態判斷。
- 因為是**相異**連通塊，所以大概率跟維護**最小、次小**有關，寫法上要注意第一點的「不要一次更新了這兩邊」。

## Dinic $O(n ^ 2 m)$

```cpp
const int N = 200, M = 5000;
const ll inf = 0x3f3f3f3f'3f3f3f3f;

struct Dinic {
    int head[N], to[M * 2], nxt[M * 2], ei;
    ll cap[M * 2], flow[M * 2];
    int q[N], dep[N];

    int s, t;

    Dinic() {
        M(flow, 0), M(head, -1);
        ei = 0;
    }

    bool bfs() {
        M(dep, 0);
        q[0] = s, dep[s] = 1;
        for (int h = 0, t = 1; h < t; ++h) {
            int i = q[h];
            for (int ei = head[i]; ~ei; ei = nxt[ei]) {
                int j = to[ei];
                if (not dep[j] and cap[ei] > flow[ei]) {
                    dep[j] = dep[i] + 1;
                    q[t++] = j;
                }
            }
        }
        return dep[t];
    }

    ll dfs(int i, ll lim) {
        if (i == t or not lim) return lim;
        ll res = 0;
        for (int ei = head[i]; ~ei; ei = nxt[ei]) {
            int j = to[ei]; ll d;
            if (dep[j] == dep[i] + 1
                    and (d = dfs(j, min(lim - res, cap[ei] - flow[ei])))) {
                res += d;
                flow[ei] += d;
                flow[ei ^ 1] -= d;
                if (res == lim) break;
            }
        }
        if (not res) dep[i] = 0;
        return res;
    }

    void add_edge(int a, int b, ll w) {
        to[ei] = b, cap[ei] = w, nxt[ei] = head[a], head[a] = ei++;
        to[ei] = a, cap[ei] = 0, nxt[ei] = head[b], head[b] = ei++;
    }

    ll maxflow(int ss, int tt) {
        s = ss, t = tt;
        ll ans = 0;
        while (bfs()) ans += dfs(s, inf);
        return ans;
    }
} dinic;
```

## 第 k 短路

```cpp
const int N = 5000, M = 2e5;
const ll inf = 0x3f3f3f3f'3f3f3f3f;

int to[M], nxt[2][M], head[2][N];
ll ew[M], dis[N];
bool ontree[M], vis[N];

int n, m, s, t, k;

// persistent randomized heap
struct node {
    ll v; int nxt;  // path
    node *le, *ri, *to;
} *roots[N];

struct item {
    ll v; int i; node *o;
    item() {}
    item(ll vv, int ii): v(vv), i(ii) {}
    item(ll vv, node *oo): v(vv), o(oo) {}
    bool operator<(const item &other) const {  // for heap/pq
        return v > other.v;
    }
};
priority_queue<item> q;

node *merge(node *a, node *b) {
    if (not a or not b) return a ? a : b;
    if (a->v > b->v) swap(a, b);
    a = new node(*a);  // persistent
    a->ri = merge(a->ri, b);
    if (rand() & 1) swap(a->le, a->ri);  // randomized
    return a;
}

void dfs(int i) {
    vis[i] = 1;
    if (i != t and dis[i] < inf) {
        for (int ei = head[0][i]; ~ei; ei = nxt[0][ei]) {
            int j = to[ei] ^ i;
            if (ontree[ei] or dis[j] >= inf) continue;
            roots[i] = merge(roots[i], new node{ew[ei] - dis[i] + dis[j], j});
        }
    }
    for (int ei = head[1][i]; ~ei; ei = nxt[1][ei]) {
        int j = to[ei] ^ i;
        if (vis[j]) continue;
        if (dis[i] + ew[ei] == dis[j]) {
            ontree[ei] = 1;
            roots[j] = roots[i];
            dfs(j);
        }
    }
}

void solve() {
    M(ontree, 0), M(vis, 0);
    M(head, -1), M(dis, 0x3f);
    cin in n in m in s in t in k;
    REP(i, 0, m) {
        int a, b; cin in a in b in ew[i];
        to[i] = a ^ b;
        nxt[0][i] = head[0][a], head[0][a] = i;
        nxt[1][i] = head[1][b], head[1][b] = i;
    }

    // Dijkstra from t
    dis[t] = 0;
    q.push({0, t});
    while (q.size()) {
        auto [v, i, _] = q.top(); q.pop();
        if (v > dis[i]) continue;
        for (int ei = head[1][i]; ~ei; ei = nxt[1][ei]) {
            int j = to[ei] ^ i;
            ll nv = v + ew[ei];
            if (nv < dis[j]) {
                dis[j] = nv;
                q.push({nv, j});
            }
        }
    }
    // shortest path tree
    dfs(t);

    // edge case k == 1
    if (--k == 0) {
        cout ot (dis[s] < inf ? dis[s] : -1) nl;
        return;
    }
    if (roots[s]) q.push({dis[s] + roots[s]->v, roots[s]});
    while (q.size()) {
        auto [v, _, o] = q.top(); q.pop();
        if (--k == 0) {
            cout ot v nl;
            return;
        }
        // add next node
        node *no = roots[o->nxt];
        if (no) q.push({v + no->v, no});

        // change last node
        if (o->le) q.push({v - o->v + o->le->v, o->le});
        if (o->ri) q.push({v - o->v + o->ri->v, o->ri});
    }
    cout ot -1 nl;
}
```

## SAM

```cpp
const int A = 26;
struct SAM {
    struct node {
        int len, link, pos; bool vis;
        ll cnt, dp;
        int nxt[A];
        int& operator[](int c) {
            return nxt[c];
        }
    } tr[N << 1];
    int last, siz;

    SAM() {
        M(tr, 0);
        tr[0] = {0, -1, -1, 0, 0};
        last = 0, siz = 1;
    }

    void add(int c) {
        int o = siz++;
        tr[o].len = tr[last].len + 1;
        tr[o].pos = tr[o].len;
        tr[o].cnt = 1;
        int p = last; last = o;
        while (~p and not tr[p][c]) {
            tr[p][c] = o;
            p = tr[p].link;
        }
        if (p == -1) {
            tr[o].link = 0;
        } else {
            int q = tr[p][c];
            if (tr[p].len + 1 == tr[q].len) {
                tr[o].link = q;
            } else {
                int clone = siz++;
                tr[clone] = tr[q];
                tr[clone].len = tr[p].len + 1;
                tr[clone].cnt = 0;
                tr[o].link = tr[q].link = clone;
                while (~p and tr[p][c] == q) {
                    tr[p][c] = clone;
                    p = tr[p].link;
                }
            }
        }
    }

    void dfs(int o) {
        if (tr[o].vis) return;
        tr[o].vis = 1;
        int pa = tr[o].link, ch;
        tr[o].dp = 1;
        REP(c, 0, A) if (ch = tr[o][c]) {
            dfs(ch);
            tr[o].dp += tr[ch].dp;
        }
        if (~pa) tr[pa].cnt += tr[o].cnt;
    }
} sam;
```

## xor basis

```cpp
template <typename T, int bl>
struct xb {
    array<T, bl> x{};
    int siz = 0;
    void add(T v) {
        if (siz == bl) return;
        for (int i = bl; i--; ) if (v >> i & 1) {
            if (not x[i]) {
                x[i] = v;
                ++siz;
                return;
            }
            v ^= x[i];
        }
    }
    bool find(T v) {
        for (int i = bl; i--; ) if (v >> i & 1) {
            if (not x[i]) return 0;
            v ^= x[i];
        }
        return 1;
    }
    T max() {
        T v = 0;
        for (int i = bl; i--; ) if (not (v >> i & 1)) {
            v ^= x[i];
        }
        return v;
    }
    T kth(T k) {
        T v = 0, cnt = (T)1 << siz;
        for (int i = bl; i--; ) if (x[i]) {
            cnt >>= 1;
            if ((cnt <= k) ^ (v >> i & 1)) v ^= x[i];
            if (cnt <= k) k -= cnt;
        }
        return v;
    }
    T rk(T v) {
        T k = 0, cnt = (T)1 << siz;
        for (int i = bl; i--; ) {
            if (x[i]) cnt >>= 1;
            if (v >> i & 1) {
                k += cnt;
                if (not x[i]) break;
                v ^= x[i];
            }
        }
        return k;
    }
};
```

```python
def slide(arr, k, func):
    n = len(arr)
    m = n - k + 1
    ans = []
    for s in range(0, m, k):
        for i in range(s + k - 1, s, -1):
            arr[i - 1] = func(arr[i - 1], arr[i])
        cur = None
        for i in range(s, min(m, s + k)):
            if i > s: cur = func(cur, arr[i + k - 1])
            res = func(arr[i], cur)
            ans.append(res)
    return ans
```
